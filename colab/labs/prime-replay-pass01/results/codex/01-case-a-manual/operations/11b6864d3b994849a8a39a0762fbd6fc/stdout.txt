def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def call(*argv: str) -> tuple[int, dict]:
    buf = io.BytesIO()

    class Out:
        buffer = buf

    old = sys.stdout
    sys.stdout = Out()
    try:
        code = pb.main([*argv, "--json"])
    finally:
        sys.stdout = old
    raw = buf.getvalue()
    return code, (json.loads(raw) if raw else {})
